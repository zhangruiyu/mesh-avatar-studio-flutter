cmake_minimum_required(VERSION 3.5)
project(mesh_avatar_studio)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(PLUGIN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_library(mesh_avatar_ffi SHARED IMPORTED)
set_target_properties(mesh_avatar_ffi PROPERTIES IMPORTED_LOCATION "${PLUGIN_ROOT}/ohos/libs/arm64-v8a/libmesh_avatar_ffi.so" IMPORTED_NO_SONAME TRUE)
add_library(avatar_bridge SHARED avatar_bridge.cpp)
target_include_directories(avatar_bridge PRIVATE "${PLUGIN_ROOT}/native/include")
target_link_libraries(avatar_bridge PUBLIC mesh_avatar_ffi libace_napi.z.so libnative_window.so libEGL.so libGLESv3.so)
